# while loop, if-else chain, and break
while (true)
{
	printchar '>'; printchar ' ';
	input grade = 999; 	    # input
	if grade == 999 break;  # break

	if grade >= 90 	# if-else chain
    	printchar 'A';
	else if grade >= 80
    	printchar 'B';
	else if grade >= 70
    	printchar 'C';
	else if grade >= 60
    	printchar 'D';
	else printchar 'F';
	println;
}
println;  
run

# recursion
func fib(n)
{
    if n <= 1 return n;
    return fib(n-1) + fib(n-2);
}
# print 20 first fibonacci numbers
for var i; i < 20; ++i { print fib(i); printchar ' '; }
run


# global inside functions acts like static keyword in C
func count()
{
    global counter = 0; # initialized once
    ++counter;
    return counter;
}
repeat 10 { print count( ); printchar ' '; }

# prime number check
printchar '>'; printchar ' ';
input n;
var isPrime = true;
if n < 2
    isPrime = false;
else {
    for var i = 2, i * i <= n, ++i
        if n % i == 0 {
            isPrime = 0;
            break;
        }
}
if isPrime printchar 'Y';
else printchar 'N';
println;
run

# factorial using block expression
printchar '>'; printchar ' ';
input n = 0;
var factorial = {
    var result = 1;
    for (var i = 1, i <= n, ++i)
        result = result * i;
    result; 	# block evaluates to result
};
print factorial; println;
run
